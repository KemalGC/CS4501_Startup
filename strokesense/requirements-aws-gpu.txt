# Dedicated paid CUDA inference worker. Do not also install CPU onnxruntime.
# Deliberately does NOT include requirements-hosted.txt: the worker never serves
# HTTP (no starlette/uvicorn/httpx), and those pins require a newer Python than
# the CUDA base image's Ubuntu 22.04 python3 (3.10), which broke the build.
# Prototype constraints; lock all packages before paid launch.
numpy>=1.24,<2.1
scipy>=1.10,<1.16
opencv-python-headless>=4.8,<5
matplotlib>=3.7,<4
# matplotlib hard-requires Pillow. 13.x (fixes 2026 TIFF/JPEG/PDF/... CVEs) needs
# Python >=3.11, which the Ubuntu 22.04 CUDA base lacks; the worker only encodes
# its own PNG charts (no untrusted image decoding). Upgrade with the base image.
pillow>=12.3.0,<13
# beautifulsoup4/soupsieve intentionally omitted: only strokesense.llm_feedback
# imports bs4, and it is not in the GPU worker's import closure.
imageio-ffmpeg>=0.5,<1
# The SHA256-pinned YOLO model declares ONNX opset 22. Runtime 1.20.2 only
# supports opset 21 and rejects it before CUDA initialization. Keep the model
# unchanged and use 1.21.1. Its exact Linux wheel reports CUDA 12.2, compatible
# with the reviewed CUDA 12.6.3/cuDNN 9 base; the image build checks its linkage.
onnxruntime-gpu==1.21.1
boto3>=1.34,<2
